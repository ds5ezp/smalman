[대시보드 셔틀 지도 On/Off · 표시 시작 시점 붙여넣기 안내 (개정 2)]

대시보드(업무포털) 프로젝트는 이 저장소에 없으므로, 아래 내용을 대시보드 프로젝트에 직접 붙여넣어 주세요.
이전 개정(1)을 붙여넣었다면 그 함수들을 아래 내용으로 교체하면 됩니다.
붙여넣은 뒤에는 대시보드를 "새 버전"으로 배포해야 반영됩니다.

운영설정 시트에 두 칸이 생깁니다(없으면 자동으로 추가됩니다).
  - '셔틀 지도 표시' : ON / OFF
  - '셔틀 표시 시작(분 전)' : 5, 10, 15, 20, 30 중 하나 (기본 10)

1) 대시보드 Code.gs : 기존 SHUTTLE_SIGNAGE 관련 함수를 모두 지우고 아래로 교체

// 사이니지 셔틀 지도 설정: 운영설정 시트의 두 칸
const SHUTTLE_SIGNAGE_LABEL = '셔틀 지도 표시';
const SHUTTLE_LEAD_LABEL = '셔틀 표시 시작(분 전)';

function readSettingValue_(label) {
  const row = readSettings_().rows.find(x => String(x.key).trim() === label);
  return row ? String(row.value).trim() : '';
}

function getShuttleSignageOn() {
  requireUser_();
  const lead = parseInt(readSettingValue_(SHUTTLE_LEAD_LABEL), 10);
  return {
    on: readSettingValue_(SHUTTLE_SIGNAGE_LABEL).toUpperCase() === 'ON',
    lead: isNaN(lead) ? 10 : Math.min(Math.max(lead, 0), 60)
  };
}

// 운영설정 한 칸을 찾아 값을 바꾸고, 없으면 맨 아래에 한 줄 추가
function upsertSettingRow_(sheet, label, value, note) {
  const labels = sheet.getRange(1, 1, Math.max(sheet.getLastRow(), 1), 1).getDisplayValues();
  const idx = labels.findIndex(r => String(r[0]).trim() === label);
  if (idx === -1) {
    sheet.appendRow([label, value, note]);
  } else {
    sheet.getRange(idx + 1, 2).setNumberFormat('@').setValue(value);
  }
}

function saveShuttleSignage(on, lead) {
  requireUser_();
  return withLock_(() => {
    const sheet = lobbySheet_('운영설정');
    const minutes = Math.min(Math.max(parseInt(lead, 10) || 10, 0), 60);
    upsertSettingRow_(sheet, SHUTTLE_SIGNAGE_LABEL, on ? 'ON' : 'OFF', '셔틀 실시간 지도를 전관 사이니지 위쪽에 띄울지 (ON/OFF)');
    upsertSettingRow_(sheet, SHUTTLE_LEAD_LABEL, String(minutes), '셔틀이 출발하기 몇 분 전부터 지도에 나올지 (5/10/15/20/30)');
    SpreadsheetApp.flush();
    log_('셔틀 지도 사이니지 표시', (on ? 'ON' : 'OFF') + ' / ' + minutes + '분 전');
    return { on: !!on, lead: minutes };
  });
}

(b) <script> 안의 기존 함수 3개(loadShuttleSignageSwitch, setShuttleSignageSwitch, toggleShuttleSignage)를 아래로 교체

    function loadShuttleSignageSwitch() {
      run('getShuttleSignageOn').then(setShuttleSignageSwitch).catch(fail);
    }

    function setShuttleSignageSwitch(s) {
      document.getElementById('shuttleSignageSwitch').checked = !!s.on;
      document.getElementById('shuttleSignageLabel').innerText = s.on ? '켜짐' : '꺼짐';
      document.getElementById('shuttleLeadSelect').value = String(s.lead);
    }

    function toggleShuttleSignage(on) {
      const lead = document.getElementById('shuttleLeadSelect').value;
      save('saveShuttleSignage', on, lead).then(s => {
        setShuttleSignageSwitch(s);
        toast(on ? '사이니지에 셔틀 지도를 띄웁니다. 약 1분 안에 반영돼요.' : '사이니지에서 셔틀 지도를 뺍니다.');
      }).catch(err => { fail(err); loadShuttleSignageSwitch(); });
    }

    function saveShuttleLead(lead) {
      const on = document.getElementById('shuttleSignageSwitch').checked;
      save('saveShuttleSignage', on, lead).then(s => {
        setShuttleSignageSwitch(s);
        toast('출발 ' + s.lead + '분 전부터 지도에 나오게 했습니다. 약 1분 안에 반영돼요.');
      }).catch(err => { fail(err); loadShuttleSignageSwitch(); });
    }
