[대시보드 셔틀 지도 On/Off 붙여넣기 안내]

대시보드(업무포털) 프로젝트는 이 저장소에 없으므로, 아래 내용을 대시보드 프로젝트에 직접 붙여넣어 주세요.
붙여넣은 뒤에는 대시보드를 "새 버전"으로 배포해야 반영됩니다.

사이니지 쪽은 이미 반영되어 있습니다. 운영설정 시트의 '셔틀 지도 표시' 칸(B열)이 ON일 때만 위쪽에 지도가 나옵니다.


=====================================================================
1) 대시보드 Code.gs 에 추가 (아무 함수 아래나 붙여넣기)
=====================================================================

// 사이니지 셔틀 지도 켜기/끄기: 운영설정 시트의 '셔틀 지도 표시' 칸 (ON/OFF)
const SHUTTLE_SIGNAGE_LABEL = '셔틀 지도 표시';

function shuttleSignageOn_() {
  const row = readSettings_().rows.find(x => String(x.key).trim() === SHUTTLE_SIGNAGE_LABEL);
  return !!row && String(row.value).trim().toUpperCase() === 'ON';
}

function getShuttleSignageOn() {
  requireUser_();
  return shuttleSignageOn_();
}

function saveShuttleSignage(on) {
  requireUser_();
  return withLock_(() => {
    const sheet = lobbySheet_('운영설정');
    const labels = sheet.getRange(1, 1, Math.max(sheet.getLastRow(), 1), 1).getDisplayValues();
    const idx = labels.findIndex(r => String(r[0]).trim() === SHUTTLE_SIGNAGE_LABEL);
    const value = on ? 'ON' : 'OFF';
    if (idx === -1) {
      sheet.appendRow([SHUTTLE_SIGNAGE_LABEL, value, '셔틀 실시간 지도를 전관 사이니지 위쪽에 띄울지 (ON/OFF)']);
    } else {
      sheet.getRange(idx + 1, 2).setNumberFormat('@').setValue(value);
    }
    SpreadsheetApp.flush();
    log_('셔틀 지도 사이니지 표시', value);
    return on;
  });
}

(b) <script> 안, function loadShuttleStatus() { 바로 위에 추가

    function loadShuttleSignageSwitch() {
      run('getShuttleSignageOn').then(setShuttleSignageSwitch).catch(fail);
    }

    function setShuttleSignageSwitch(on) {
      document.getElementById('shuttleSignageSwitch').checked = on;
      document.getElementById('shuttleSignageLabel').innerText = on ? '켜짐' : '꺼짐';
    }

    function toggleShuttleSignage(on) {
      save('saveShuttleSignage', on).then(v => {
        setShuttleSignageSwitch(v);
        toast(on ? '사이니지에 셔틀 지도를 띄웁니다. 약 1분 안에 반영돼요.' : '사이니지에서 셔틀 지도를 뺍니다.');
      }).catch(err => { fail(err); loadShuttleSignageSwitch(); });
    }
